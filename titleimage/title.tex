% \WindowBar{node}{title}: the title bar of a window
\newcommand\WindowBar[2]{%
   \fill[Bar] ([yshift=-3.5mm]#1.north west) [rounded corners=2pt] -- (#1.north west) -- (#1.north east) [sharp corners] -- ([yshift=-3.5mm]#1.north east) -- cycle;
   \node[font=\tiny\sbseries,white,anchor=west] at([shift={(1.5mm,-1.75mm)}]#1.north west) {#2};
   \node[font=\tiny,white,anchor=east] at([shift={(-1.5mm,-1.75mm)}]#1.north east) {\faCircle\,\faCircle\,\faCircle};
}
